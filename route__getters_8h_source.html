41 const size_t n = route.route_elements.size();
42 size_t start_idx = incl_undershoot ? 0 : route.starting_route_element_idx;

55 const size_t n = route.route_elements.size();
56 size_t start_idx = route.current_route_element_idx;

64 return std::vector<RouteElement>(route.route_elements.begin() + start_idx,

76inline size_t getRouteElementIdxClosestToS(const Route& route, const double s) {
77 if (route.route_elements.size() < 2) {
78 throw std::runtime_error("Not enough route elements to calculate difference");
79 }
80
81 double min_difference = 2 * std::abs(route.route_elements[1].s - route.route_elements[0].s);
82 auto it = std::min_element(route.route_elements.begin(), route.route_elements.end(),
83 [s](const RouteElement& a, const RouteElement& b) {
84 return std::abs(a.s - s) < std::abs(b.s - s);
85 });
86
87 if (it == route.route_elements.end() || std::abs(it->s - s) >= min_difference) {
88 throw std::runtime_error("No route element found within acceptable difference");
89 }
90
91 return std::distance(route.route_elements.begin(), it);
92}

98inline double getWidthOfLaneElement(const LaneElement& lane_element) {
99 double dx = lane_element.left_boundary.point.x - lane_element.right_boundary.point.x;
100 double dy = lane_element.left_boundary.point.y - lane_element.right_boundary.point.y;
101 return std::sqrt(dx * dx + dy * dy);

118inline double estimateRemainingTime(const Route& route,
119 const double reference_speed_mps = DEFAULT_REFERENCE_SPEED_MPS,
120 const double calibration_factor = REMAINING_TIME_ESTIMATION_FACTOR) {
121 const std::vector<RouteElement> remaining_route_elements =
122 getRemainingRouteElements(route, route.destination_route_element_idx == Route::INVALID_ROUTE_ELEMENT_IDX);
123 if (remaining_route_elements.size() < 2 || reference_speed_mps <= 0.0 || calibration_factor <= 0.0) {
124 return 0.0;
125 }
126 double remaining_time = 0.0;
127 for (size_t i = 0; i + 1 < remaining_route_elements.size(); ++i) {
128 const auto& route_element = remaining_route_elements[i];
129 const auto& next_route_element = remaining_route_elements[i + 1];
130 const uint8_t speed_limit_kmh = getSuggestedLaneElement(route_element).speed_limit;
131 double speed_mps = static_cast<double>(speed_limit_kmh) / 3.6;
132 if (speed_limit_kmh == LaneElement::SPEED_LIMIT_UNKNOWN) {
133 speed_mps = reference_speed_mps;
134 }
135 if (speed_limit_kmh == LaneElement::SPEED_LIMIT_UNLIMITED) {
136 speed_mps = UNLIMITED_SPEED_MPS;
137 }
138 remaining_time += (next_route_element.s - route_element.s) / speed_mps;
139 }
140 return calibration_factor * remaining_time;
141}

212inline bool hasAdjacentLane(const RouteElement& route_element, const size_t lane_idx, const int lane_diff_idx){
213 int adjacent_lane_idx = lane_idx + lane_diff_idx;
214 if (lane_idx >= route_element.lane_elements.size()) {
215 throw std::invalid_argument("Lane index out of range: " + std::to_string(lane_idx));
216 }
217
218 if (adjacent_lane_idx >= static_cast<int>(route_element.lane_elements.size()) || adjacent_lane_idx < 0) {
219 return false;
220 }
221 return true;

240inline int getLaneChangeDirection(const RouteElement& route_element, const RouteElement& following_route_element) {
241 if (!route_element.will_change_suggested_lane) {
242 throw std::invalid_argument("Route element does not indicate a change of suggested lane");
243 }
244 size_t current_lane_idx = route_element.suggested_lane_idx;
245 size_t target_lane_idx;
246 if (auto result = getPrecedingLaneElementIdx(following_route_element.suggested_lane_idx, route_element)) {
247 target_lane_idx = *result;
248 } else {
249 throw std::invalid_argument("No preceding lane element found for route element with suggested lane " + std::to_string(route_element.suggested_lane_idx));
250 }
251 int lane_change_direction = target_lane_idx - current_lane_idx;
252 return lane_change_direction;
253}

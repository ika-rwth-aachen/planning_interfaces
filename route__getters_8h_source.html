92inline std::vector<RouteElement*> getRemainingRouteElementsAsPointers(Route& route, const bool incl_overshoot = false) {
93 const size_t n = route.route_elements.size();
94 if (route.current_route_element_idx >= n) {
95 return {};
96 }
97
98 const size_t start_idx = static_cast<size_t>(route.current_route_element_idx);
99 const size_t end_idx = !incl_overshoot && route.destination_route_element_idx < n
100 ? static_cast<size_t>(route.destination_route_element_idx) + 1
101 : n;
102 if (start_idx >= end_idx) {
103 return {};
104 }
105
106 std::vector<RouteElement*> result;
107 result.reserve(end_idx - start_idx);
108 for (size_t idx = start_idx; idx < end_idx; ++idx) {
109 result.push_back(&route.route_elements[idx]);
110 }
111 return result;
112}

133inline size_t getRouteElementIdxClosestToS(const Route& route, const double s,
134 const double max_distance = std::numeric_limits<double>::infinity()) {
135 if (max_distance < 0.0 || std::isnan(max_distance)) {
136 throw std::invalid_argument("Maximum distance must be non-negative");
137 }
138 if (route.route_elements.empty()) {
139 throw std::runtime_error("Cannot find a route element in an empty route");
140 }
141
142 auto it = std::min_element(route.route_elements.begin(), route.route_elements.end(),
143 [s](const RouteElement& a, const RouteElement& b) {
144 return std::abs(a.s - s) < std::abs(b.s - s);
145 });
146 if (std::abs(it->s - s) > max_distance) {
147 throw std::runtime_error("No route element found within maximum distance");
148 }
149 return std::distance(route.route_elements.begin(), it);
150}

177inline double getWidthOfLaneElement(const LaneElement& lane_element) {
178 double dx = lane_element.left_boundary.point.x - lane_element.right_boundary.point.x;
179 double dy = lane_element.left_boundary.point.y - lane_element.right_boundary.point.y;
180 return std::sqrt(dx * dx + dy * dy);
181}

213inline double estimateRemainingTime(const Route& route,
214 const double reference_speed_mps = DEFAULT_REFERENCE_SPEED_MPS,
215 const double calibration_factor = REMAINING_TIME_ESTIMATION_FACTOR) {
216 const std::vector<RouteElement> remaining_route_elements =
217 getRemainingRouteElements(route);
218 if (remaining_route_elements.size() < 2 || reference_speed_mps <= 0.0 || calibration_factor <= 0.0) {
219 return 0.0;
220 }
221 double remaining_time = 0.0;
222 for (size_t i = 0; i + 1 < remaining_route_elements.size(); ++i) {
223 const auto& route_element = remaining_route_elements[i];
224 const auto& next_route_element = remaining_route_elements[i + 1];
225 const uint8_t speed_limit_kmh = getSuggestedLaneElement(route_element).speed_limit;
226 double speed_mps = static_cast<double>(speed_limit_kmh) / 3.6;
227 if (speed_limit_kmh == LaneElement::SPEED_LIMIT_UNKNOWN) {
228 speed_mps = reference_speed_mps;
229 }
230 if (speed_limit_kmh == LaneElement::SPEED_LIMIT_UNLIMITED) {
231 speed_mps = UNLIMITED_SPEED_MPS;
232 }
233 remaining_time += (next_route_element.s - route_element.s) / speed_mps;
234 }
235 return calibration_factor * remaining_time;
236}

382inline bool hasAdjacentLane(const RouteElement& route_element, const size_t lane_idx, const int lane_diff_idx){
383 int adjacent_lane_idx = lane_idx + lane_diff_idx;
384 if (lane_idx >= route_element.lane_elements.size()) {
385 throw std::invalid_argument("Lane index out of range: " + std::to_string(lane_idx));
386 }
387
388 if (adjacent_lane_idx >= static_cast<int>(route_element.lane_elements.size()) || adjacent_lane_idx < 0) {
389 return false;
390 }

410inline int getLaneChangeDirection(const RouteElement& route_element, const RouteElement& following_route_element) {
411 if (!route_element.will_change_suggested_lane) {
412 throw std::invalid_argument("Route element does not indicate a change of suggested lane");
413 }
414 size_t current_lane_idx = route_element.suggested_lane_idx;
415 size_t target_lane_idx;
416 if (auto result = getPrecedingLaneElementIdx(following_route_element.suggested_lane_idx, route_element)) {
417 target_lane_idx = *result;
418 } else {
419 throw std::invalid_argument("No preceding lane element found for route element with suggested lane " + std::to_string(route_element.suggested_lane_idx));
420 }
421 int lane_change_direction = target_lane_idx - current_lane_idx;
422 return lane_change_direction;
423}
